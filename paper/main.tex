\begin{abstract}
Conjecture~133 of \emph{Written on the Wall II} states that every connected graph $G$ on at least two vertices with no $4$-cycle subgraph satisfies $\pp(G)\ge r(G)+\lr(G)$. We call this statement REAL-133. Here $\pp(G)$ is the largest order of an induced path, $r(G)$ is the radius, and $\lr(G)=S(G)/n$ is the average of $\alpha(G[N(v)])$.

For connected strong $C_4$-free $P_6$-free graphs we prove $S\le 3n$, with equality only for the Petersen graph. In that class, $\lr>5/2$ holds only for the Petersen graph and the graph obtained by deleting one vertex, with values $3$ and $8/3$. We prove REAL-133 when $\lr\le 3$, when $r\le 2$ and $\lr\le 4$, when $r=4$ and $\lr\le 4$, and when $r\ge 5$ and $\lr\le 4$. The case $r=4$ and $\pp=7$ gives the stronger bound $\lr\le 2$. The only unsettled case with $\lr\le 4$ is $r=3$, $\pp=6$ and $3<\lr\le 4$.

A theorem of Erd\H{o}s, Saks and S\'os reduces any counterexample to a radius-free bound $\GEN_K$ or a radius-extremal bound $\RAD(r)$. We prove the rooted bound of order~$3$, and the rooted bound of every order for connected strong $C_4$-free graphs whose blocks are bipartite or cycles. Explicit examples limit several strengthenings. The bounds $\GEN_K$ for $K\ge 3$, $\RAD(r)$ for $r\ge 5$, the rooted bound of order at least~$4$ outside that block class, unrestricted $\GEN_6$, and REAL-133 remain open.
\end{abstract}

\section{Introduction}

All graphs are finite and simple. For a graph $G$ with $n\ge 2$ vertices let $\pp(G)$ be the largest number of vertices of an induced path, $r(G)$ the radius, and
\[
\lr(G)=\frac1n\sum_{v\in V(G)}\alpha\bigl(G[N(v)]\bigr),
\]
where $\alpha$ is the independence number and $N(v)$ the open neighbourhood. Write $c_{C_4}(G)=1$ if $G$ contains no $4$-cycle as a subgraph and $c_{C_4}(G)=0$ otherwise.

\begin{conjecture}[Graffiti/WOWII Conjecture 133]\label{conj:133}
Every connected graph $G$ on at least two vertices satisfies
\[
r(G)+\lr(G)^{\,c_{C_4}(G)}\le \pp(G).
\]
\end{conjecture}

The conjecture is number~133 in E.~DeLaVina's list \emph{Written on the Wall II} of conjectures of the program Graffiti.pc~\cite{wowii}. We follow the formal statement in the Formal Conjectures repository~\cite{fc133}, which reads \texttt{(rad : $\mathbb{R}$) + l G \^{} cC4 $\le$ (path G : $\mathbb{R}$)}. Earlier versions of that file had $\lfloor \lr(G)\rfloor$ in place of $\lr(G)$. Pull request \#5727 of the repository~\cite{fcpr} removed the floor. We work with the real-valued statement and call it REAL-133.

If $G$ contains a $4$-cycle, the conjecture reads $\pp(G)\ge r(G)+1$. A shortest path between two vertices at distance $\operatorname{diam}(G)\ge r(G)$ is induced, so the inequality holds. Only graphs with no $4$-cycle subgraph remain. We call them \emph{strong $C_4$-free}. Since $\pp$ and $r$ are integers, REAL-133 for such a graph is equivalent to
\begin{equation}\label{eq:target}
\pp(G)-r(G)\ \ge\ \lceil \lr(G)\rceil .
\end{equation}

H.~Chen~\cite{chen} has studied mainly the floor version $\pp\ge r+\lfloor\lr\rfloor$. That preprint describes itself as not peer-reviewed, and we have not checked its proofs. For the real-valued form it records an exhaustive check on strong $C_4$-free graphs with at most $13$ vertices, and its Theorem~6.7 states that $\operatorname{diam}(G)=2$ and $\lr>3$ imply $\pp\ge 6$. This overlaps the case $r=2$, $3<\lr\le 4$ of our Theorem~\ref{thm:A} for graphs of diameter~$2$. Its Lemma~7.18 contains the trace classification in the first part of our Lemma~\ref{lem:c6}. Our proofs are independent of~\cite{chen}.

The arguments below are written in full. Section~\ref{sec:small} gives the small-path mass bounds and their extremal graphs. Section~\ref{sec:r4} treats radius four, including the case $\pp=7$. Section~\ref{sec:band} is the reduction and a table of what remains. Section~\ref{sec:blocks} is the rooted theorem for every index in the block class. Several inequalities studied by finite search are not theorems here: the repository does not contain a checker, an input list, and a completeness argument for them. They are listed in Section~\ref{sec:notclaimed} and are not used as hypotheses.

\section{Preliminaries}\label{sec:prelim}

For $v\in V(G)$ let $d(v)$ be its degree and $t(v)$ the number of triangles containing $v$. Write $\lambda(v)=\lambda_G(v)=\alpha(G[N(v)])$, $S(G)=\sum_v\lambda(v)$, $e(G)$ for the number of edges and $T(G)$ for the number of triangles.

\begin{lemma}\label{lem:basic}
Let $G$ be strong $C_4$-free. Then:
\begin{enumerate}
\item two distinct vertices have at most one common neighbour;
\item for every $v$, $G[N(v)]$ is a matching together with isolated vertices, so $\lambda(v)$ equals the number of components of $G[N(v)]$ and $\lambda(v)=d(v)-t(v)$;
\item $S(G)=2e(G)-3T(G)$.
\end{enumerate}
\end{lemma}

\begin{proof}
(1) Two common neighbours $w,w'$ of $x,y$ give the $4$-cycle $xwyw'$. (2) A vertex of $N(v)$ with two neighbours in $N(v)$ would have two common neighbours with $v$. So the components are single vertices and edges, each edge being a triangle at $v$. (3) Sum $d(v)-t(v)$ over $v$.
\end{proof}

The components of $G[N(v)]$ are the \emph{units} of $v$. For a vertex set $U$, $G-U$ is the subgraph induced by $V\setminus U$, and $D_G(U)=S(G)-S(G-U)$ is the \emph{deletion loss} of $U$. We write $L(v)=D_G(\{v\})$.

\begin{lemma}\label{lem:loss}
Let $G$ be strong $C_4$-free and $\emptyset\ne U\subseteq V(G)$. Then
\[
D_G(U)=\sum_{u\in U}\lambda(u)+\sum_{x\notin U}\#\{\text{units of }x\text{ contained in }U\}.
\]
In particular $L(v)=2\lambda(v)-t(v)=2d(v)-3t(v)$.
\end{lemma}

\begin{proof}
For $x\notin U$, deleting $U$ removes exactly the units of $x$ that lie inside $U$; a unit that is an edge with one end in $U$ survives as a vertex. For $U=\{v\}$: the $2t(v)$ neighbours of $v$ in triangles with $v$ keep their value, and each of the other $d(v)-2t(v)$ neighbours loses one. The total is $\lambda(v)+d(v)-2t(v)=2\lambda(v)-t(v)$.
\end{proof}

\begin{lemma}\label{lem:cut}
Let $v$ be a cut vertex of $G$. Split the components of $G-v$ into two nonempty groups, and let $G_1,G_2$ be the subgraphs induced by the two groups together with $v$. Then $S(G)=S(G_1)+S(G_2)$.
\end{lemma}

\begin{proof}
$G[N(v)]$ has no edges between the groups, so $\lambda_G(v)=\lambda_{G_1}(v)+\lambda_{G_2}(v)$. Every other vertex keeps its neighbourhood.
\end{proof}

We use three published results.

\begin{theorem}[Erd\H{o}s, Saks and S\'os~\cite{ess}]\label{thm:ess}
Every connected graph $G$ has an induced path on at least $2r(G)-1$ vertices.
\end{theorem}

\begin{theorem}[Camby and Schaudt~{\cite[Theorem~5]{cs}}]\label{thm:cs}
Let $k\ge 4$ and let $G$ be a connected graph with no induced path on $k$ vertices. Then $G$ has a connected dominating set $X$ such that $G[X]$ has no induced path on $k-2$ vertices or $G[X]$ is isomorphic to the cycle $C_k$.
\end{theorem}

\begin{theorem}[Seinsche~\cite{seinsche}]\label{thm:sein}
If a graph on at least two vertices has no induced path on four vertices, then it or its complement is disconnected.
\end{theorem}

Theorem~\ref{thm:cs} is the implication (i)$\Rightarrow$(ii) of~\cite[Theorem~5]{cs} applied with $H=G$. We write $P_k$ for the path on $k$ vertices, and ``$G$ is $P_k$-free'' means that $G$ has no induced $P_k$, that is, $\pp(G)\le k-1$. A connected graph of radius $r$ has at least $2r$ vertices.

\begin{lemma}\label{lem:join}
Let $H$ be a connected strong $C_4$-free $P_4$-free graph on at least two vertices. Then $H$ has a vertex adjacent to all other vertices.
\end{lemma}

\begin{proof}
By Theorem~\ref{thm:sein}, $H$ is the join of two nonempty vertex sets $A$ and $B$. If $|A|,|B|\ge 2$, then $a,a'\in A$ and $b,b'\in B$ span the $4$-cycle $aba'b'$. So one side is a single vertex, and it is adjacent to everything.
\end{proof}

\section{Small-path mass bounds}\label{sec:small}

\begin{lemma}\label{lem:r1}
If $G$ is connected and strong $C_4$-free with $r(G)=1$, then $S(G)=2(n-1)-m$, where $m$ is the number of edges of $G[N(u)]$ for a dominating vertex $u$. Hence $\lr(G)<2$, and~\eqref{eq:target} holds.
\end{lemma}

\begin{proof}
Let $v\ne u$. By Lemma~\ref{lem:basic}(1), $v$ has at most one neighbour other than $u$, and such a neighbour lies in $N(u)$. So $G[N(v)]$ is a clique and $\lambda(v)=1$; also $\lambda(u)=(n-1)-m$. Hence $S=2(n-1)-m<2n$ and $\lceil\lr\rceil\le 2$. If $G$ is complete then $G\in\{K_2,K_3\}$, $\lr=1$ and $\pp=2=r+1$. Otherwise two nonadjacent vertices of $N(u)$ together with $u$ form an induced $P_3$, so $\pp\ge 3\ge r+\lceil\lr\rceil$.
\end{proof}

\begin{lemma}\label{lem:r2}
If $G$ is connected and strong $C_4$-free with $r(G)\ge 2$, then $\pp(G)\ge 4$.
\end{lemma}

\begin{proof}
If $G$ were $P_4$-free, Lemma~\ref{lem:join} would give a dominating vertex, so $r(G)\le 1$.
\end{proof}

\begin{lemma}\label{lem:p5}
If $G$ is connected, strong $C_4$-free and $P_5$-free, then $S(G)\le 2n-h$, where $h$ is the number of vertices $v$ with $\lambda(v)\ge 3$. In particular $S(G)\le 2n$.
\end{lemma}

\begin{proof}
Call $v$ \emph{simplicial} if $\lambda(v)=1$ and \emph{high} if $\lambda(v)\ge 3$. For $x\in N(v)$ put $X_x=N(x)\setminus N[v]$; then $N(x)\subseteq\{v\}\cup\{\bar x\}\cup X_x$, where $\bar x$ is the partner of $x$ if $x$ lies in an edge unit of $v$.

\emph{Key fact.} Let $x,y$ lie in different units of $v$. A common vertex of $X_x$ and $X_y$ would give the $4$-cycle $vxwy$, so $X_x\cap X_y=\emptyset$. If $x'\in X_x$ and $y'\in X_y$ are nonadjacent, then $x'xvyy'$ is an induced $P_5$: $x\not\sim y$ because they lie in different units, $x\not\sim y'$ and $y\not\sim x'$ by disjointness, and $v$ is adjacent to neither $x'$ nor $y'$. So $X_x$ is complete to $X_y$. If both are nonempty and $|X_x|\ge 2$, a vertex of $X_y$ has two neighbours in $X_x$, which are also neighbours of $x$, a contradiction to Lemma~\ref{lem:basic}(1). Hence if both are nonempty, then $|X_x|=|X_y|=1$.

\emph{Main step.} For each high $v$ we find $\lambda(v)-1$ simplicial vertices whose only high neighbour is $v$, one in each of $\lambda(v)-1$ distinct units of $v$.

Case 1: some $u\in N(v)$ has $|X_u|\ge 2$. By the key fact $X_y=\emptyset$ for every $y$ outside the unit of $u$. Then $N(y)\subseteq\{v,\bar y\}$, which is a clique, so $y$ is simplicial; so is $\bar y$. Hence $v$ is the only high neighbour of $y$, and the $\lambda(v)-1$ units other than that of $u$ supply the vertices.

Case 2: $|X_x|\le 1$ for all $x\in N(v)$. Then $\lambda(x)\le 2$ for $x\in N(v)$. Call a unit \emph{active} if one of its vertices $x$ has $X_x\ne\emptyset$. Suppose two units are active, with $y'\in X_y$ and $z'\in X_z$, and let $x$ lie in a third unit. By the key fact $y'\sim z'$, and $xvyy'z'$ is an induced $P_5$: $x\not\sim y$ (different units), $x\not\sim y',z'$ (disjointness), $v\not\sim y',z'$, and $y\not\sim z'$ (disjointness). As $\lambda(v)\ge 3$, a third unit exists, so at most one unit is active. Every vertex of an inactive unit is simplicial and its only high neighbour is $v$. This gives at least $\lambda(v)-1$ units.

\emph{Counting.} The chosen vertices for distinct high vertices are distinct, since each has a unique high neighbour. So $\sum_{v\text{ high}}(\lambda(v)-1)\le\#\{\text{simplicial vertices}\}$, and
\[
S=\sum_v\lambda(v)\le 2n-\#\{\text{simplicial}\}+\sum_{v\text{ high}}(\lambda(v)-2)\le 2n-h.\qedhere
\]
\end{proof}

\begin{lemma}\label{lem:c6}
Let $G$ be connected, strong $C_4$-free and $P_6$-free with a dominating induced cycle $C=v_0v_1\cdots v_5$, and let $W=V(G)\setminus V(C)$. Then every $w\in W$ is adjacent to exactly one antipodal pair $\{v_i,v_{i+3}\}$ and to no other vertex of $C$, distinct vertices of $W$ use distinct pairs, and $W$ is independent. Hence $|W|\le 3$, $S(G)=12+4|W|<3n$, and $r(G)\le 2$ if $W\ne\emptyset$.
\end{lemma}

\begin{proof}
Indices are modulo $6$. Let $w\in W$; it has a neighbour on $C$. If $w$ has exactly one neighbour $v_i$, then $wv_iv_{i+1}v_{i+2}v_{i+3}v_{i+4}$ is an induced $P_6$. If $N(w)\cap C=\{v_i,v_{i+1}\}$, then $wv_{i+1}v_{i+2}v_{i+3}v_{i+4}v_{i+5}$ is an induced $P_6$. Two neighbours at distance $2$ on $C$ give a $4$-cycle. A set of at least three vertices of $C_6$ contains two vertices at distance $2$. So $N(w)\cap C$ is an antipodal pair. Two vertices of $W$ with the same pair give two common neighbours. If $w\sim w'$ with distinct pairs, some $v_i\in N(w)$ and $v_j\in N(w')$ are adjacent on $C$, and $wv_iv_jw'$ is a $4$-cycle. There are no triangles, $e(G)=6+2|W|$, and Lemma~\ref{lem:basic}(3) gives $S=12+4|W|$. Finally, if $w$ uses $\{v_i,v_{i+3}\}$, then every vertex of $C$ is within distance $2$ of $v_i$ (the vertex $v_{i+3}$ through $w$), and each $w'\in W$ has a neighbour in $\{v_{i-1},v_i,v_{i+1}\}$, because each of the three antipodal pairs meets this set. So $v_i$ has eccentricity $2$.
\end{proof}

The same trace analysis applies to any vertex adjacent to an induced $6$-cycle in a connected strong $C_4$-free $P_6$-free graph, whether or not the cycle dominates. We use that form in Proposition~\ref{prop:gap}.

\begin{lemma}\label{lem:ecc2}
Let $G$ be connected, strong $C_4$-free and $P_6$-free. Then either $G$ has a vertex of eccentricity at most $2$, or $G=C_6$.
\end{lemma}

\begin{proof}
Apply Theorem~\ref{thm:cs} with $k=6$ to get a connected dominating set $X$. If $G[X]\cong C_6$, Lemma~\ref{lem:c6} gives $G=C_6$ or a vertex of eccentricity $2$. Otherwise $G[X]$ is $P_4$-free. If $|X|=1$, the vertex in $X$ dominates $G$. If $|X|\ge 2$, Lemma~\ref{lem:join} gives $u\in X$ adjacent to all of $X\setminus\{u\}$, and since $X$ dominates $G$, every vertex is within distance $2$ of $u$.
\end{proof}

\begin{lemma}\label{lem:p6}
If $G$ is connected, strong $C_4$-free and $P_6$-free, then $S(G)\le 3n$, with equality if and only if $G$ is the Petersen graph.
\end{lemma}

\begin{proof}
If $G=C_6$ then $S=12<18$. Otherwise, by Lemma~\ref{lem:ecc2}, fix $u$ of eccentricity at most $2$. Put $N_1=N(u)$, $N_2=V\setminus N[u]$, and let $m_1$ be the number of edges of $G[N_1]$. By Lemma~\ref{lem:basic}(1) each $c\in N_2$ has exactly one neighbour in $N_1$, its \emph{parent}. For $a\in N_1$ let $C_a$ be the set of its children. An edge $cd$ of $G[N_2]$ with distinct parents is a \emph{cross edge}. Let $E'$ be the set of cross edges, and let $D_c$ be the set of cross neighbours of $c$.

\emph{Local values.} $\lambda(u)=|N_1|-m_1$. For $a\in N_1$ with partner $\bar a$ (if any), no child of $a$ is adjacent to $u$ or $\bar a$, so the units of $a$ are $\{u,\bar a\}$ (or $\{u\}$) together with the components of $G[C_a]$; hence $\lambda(a)=1+\alpha(G[C_a])$. For $c\in C_a$: $c$ has at most one neighbour $s\in C_a$ (two would be common neighbours of $a$ and $c$), the vertices of $D_c$ are not adjacent to $a$, and $s$ is adjacent to no $d\in D_c$ (else $c,a,s,d$ is a $4$-cycle). So $\lambda(c)=1+\alpha(G[D_c])$. Using $\sum_a|C_a|=|N_2|$ and $n=1+|N_1|+|N_2|$ we get
\begin{equation}\label{eq:p6id}
3n-S=3+|N_1|+m_1+\sum_a\bigl(|C_a|-\alpha(G[C_a])\bigr)+|N_2|-\sum_{c}\alpha(G[D_c])\ \ge\ 3+|N_1|+|N_2|-2|E'|.
\end{equation}
So it suffices to show $2|E'|\le 3+|N_1|+|N_2|$.

\emph{Fact 1.} Let $cd\in E'$ with $c\in C_y$ and $d\in C_z$. Then $y\not\sim z$, else $ycdz$ is a $4$-cycle. Let $y'\in N_1\setminus\{y,z\}$ and $x\in C_{y'}$. Then $x$ is adjacent to $c$ or $d$. Indeed, otherwise: if $y'\not\sim y$, then $xy'uycd$ is an induced $P_6$; if $y'\sim y$, then $y'\not\sim z$ (since $G[N_1]$ is a matching) and $xy'uzdc$ is an induced $P_6$. Two children of $y'$ adjacent to the same vertex of $\{c,d\}$ would be two common neighbours of $y'$ and that vertex. So $|C_{y'}|\le 2$.

\emph{Fact 2.} If $E'\ne\emptyset$, at most three vertices of $N_1$ have children. Suppose $cd\in E'$ has parents $y,z$, and $y_1,y_2\in N_1\setminus\{y,z\}$ have children $x_1,x_2$. Let $H=G[\{c,d,x_1,x_2\}]$. These four vertices have distinct parents, so every edge of $H$ is a cross edge, and by Fact~1 each of the other two vertices is adjacent to an endpoint of that edge. Hence $H$ has no induced $K_2\cup K_1$. So the complement of $H$ has no induced $P_3$, that is, it is a disjoint union of cliques, and $H$ is complete multipartite. A complete multipartite graph on four vertices with at least one edge and no $4$-cycle subgraph is $K_{1,3}$. Let its centre be $c^*$ and its leaves $a,b,d^*$, with parents $a_0,b_0,d_0$. Since $G[N_1]$ is a matching, two of these parents are nonadjacent, say $a_0\not\sim b_0$. Then $a_0,u,b_0,b,c^*,d^*$ is an induced $P_6$: the leaves are pairwise nonadjacent, and each child is adjacent only to its own parent in $N_1$. This is a contradiction.

\emph{Counting.} A child has at most one cross neighbour in each other class $C_{y'}$ (two would be common neighbours with $y'$), so its number of cross neighbours is at most $k-1$, where $k\le 3$ is the number of vertices of $N_1$ with children. If $E'=\emptyset$ we are done. If $k=2$, every child has at most one cross neighbour, and $2|E'|\le|N_2|$. Let $k=3$, with classes $C_1,C_2,C_3$. If a child in $C_i$ has two cross neighbours, in $C_j$ and $C_l$, then Fact~1 applied to these two cross edges gives $|C_j|,|C_l|\le 2$. Distinct such children of $C_i$ have distinct neighbours in $C_j$ (else two common neighbours with $y_i$), so $C_i$ contains at most two children with two cross neighbours, and there are at most six in total. Hence $2|E'|\le|N_2|+6\le|N_2|+3+|N_1|$, because $|N_1|\ge k=3$.

\emph{Equality.} The Petersen graph is $P_6$-free: if $v_0\cdots v_5$ were an induced path, then $v_0$ would have a common neighbour with each of $v_3,v_4,v_5$ (diameter $2$); these lie off the path (the only path neighbour of $v_0$ is $v_1$, which is not adjacent to them) and are distinct (girth $5$), but $v_0$ has only two neighbours off the path. Also $\lambda\equiv 3$, so $S=30=3n$. Conversely, suppose $S=3n$. Then $G\ne C_6$; the dominating-$C_6$ case is excluded because there $S<3n$ (Lemma~\ref{lem:c6}); and $E'\ne\emptyset$, because otherwise \eqref{eq:p6id} gives $3n-S\ge 3+|N_1|+|N_2|>0$. If $k=2$ then $3n-S\ge 3+|N_1|>0$. So $k=3$, and equality throughout forces: $|N_1|=3$; $m_1=0$; no edges inside any $C_a$; every $D_c$ is independent; and exactly six children have two cross neighbours, two in each class. By Fact~1 every class has at most two children, so every class has exactly two, and every child has exactly two cross neighbours. The cross edges then form a $2$-regular graph on the six children with no triangle (each $D_c$ is independent), that is, a $6$-cycle. Two children of the same parent at distance $2$ on this cycle would have two common neighbours, and they are not adjacent, so they are antipodal. This determines $G$ up to isomorphism, and the Petersen graph has exactly this structure: in it, the six vertices at distance $2$ from a vertex induce a $6$-cycle.
\end{proof}

\begin{proposition}\label{prop:gap}
Let $G$ be connected, strong $C_4$-free and $P_6$-free, of order at least two. If $\lr(G)>5/2$, then $G$ is the Petersen graph $P$ or the graph $P-v$ obtained by deleting one vertex of $P$. These graphs have $\lr(P)=3$ and $\lr(P-v)=8/3$. Every graph in the class other than $P$ satisfies $\lr\le 8/3$, with equality precisely for $P-v$.
\end{proposition}

\begin{proof}
Throughout, $S=2e-3T$ by Lemma~\ref{lem:basic}. A chordless cycle of length at least seven contains an induced $P_6$. Thus the only possible holes have lengths five and six.

\emph{Case 1: an induced six-cycle.}
Label its vertices $c_0,\dots,c_5$, indices modulo~$6$, and write $A$ for the set of vertices at distance one from the cycle. The trace argument in the proof of Lemma~\ref{lem:c6} shows that each $a\in A$ meets the cycle in one antipodal pair. Distinct vertices of $A$ use distinct pairs, and $A$ is an independent set: an edge of $A$ joins distinct pairs, and some contact of one end is adjacent on the cycle to some contact of the other, giving a $4$-cycle. Hence $k=|A|\le 3$.

Every vertex $x$ at distance two from the cycle is adjacent to every vertex of $A$. Such an $x$ has at least one neighbour in $A$. If $x\sim a$ and $x\not\sim b$ for $a,b\in A$, choose contacts so that $c_i\in N(a)$ and $c_{i+2}\in N(b)$. This is possible: the pairs are distinct, so some contact of $b$ lies at cycle distance two from some contact of $a$, and we orient the cycle from that contact of $a$ toward that contact of $b$. The path $x,a,c_i,c_{i+1},c_{i+2},b$ is then an induced $P_6$. Indeed $x$ meets the cycle in nothing, $a$ meets the cycle only in its antipodal pair (neither $c_{i+1}$ nor $c_{i+2}$), $b$ meets the cycle only in its antipodal pair (so $b\sim c_{i+2}$ and $b\not\sim c_i,c_{i+1}$), and $a\not\sim b$.

There is no vertex at distance three or more from the cycle. A shortest path $z,x,a,c_i$ from such a vertex down to the cycle, extended by $c_{i+1},c_{i+2}$, is an induced $P_6$: the vertex $z$ has no neighbour in $A$ or on the cycle, and $x$ has no neighbour on the cycle.

Let $h$ be the number of vertices at distance two. If $k=0$, then $A$ is empty, so the cycle has no neighbour and $G=C_6$. If $k=1$, every distance-two vertex is adjacent to the unique vertex $a\in A$. The distance-two vertices induce a matching together with isolated vertices: a vertex of that layer with two neighbours in the layer would have two common neighbours with $a$. Writing $m$ for the number of edges of the matching gives $n=7+h$, $e=8+h+m$ and $T=m$, so $S=16+2h-m$. Then
\[
5n-2S=5(7+h)-2(16+2h-m)=3+h+2m>0,
\]
hence $\lr<5/2$.

If $k\in\{2,3\}$, then $h\le 1$: two vertices of the second layer would both be adjacent to two vertices of $A$. There are no triangles. The second-layer vertex, if it exists, is adjacent to every vertex of $A$ and to nothing on the cycle, $A$ is independent, and each vertex of $A$ contributes two edges to the cycle. Thus $n=6+k+h$ and $S=2(6+2k+kh)$. The four possibilities are
\begin{center}
\begin{tabular}{@{}ccccc@{}}
\toprule
$(k,h)$ & $(2,0)$ & $(2,1)$ & $(3,0)$ & $(3,1)$\\
\midrule
$S/n$ & $5/2$ & $8/3$ & $8/3$ & $3$\\
\bottomrule
\end{tabular}
\end{center}
The graph with $(k,h)=(3,1)$ is the equality case of Lemma~\ref{lem:p6}: the second-layer vertex has three pairwise nonadjacent neighbours, each with two children, and those six children induce a $6$-cycle on which children of the same parent are antipodal. That graph is the Petersen graph $P$. Deleting the second-layer vertex leaves $(3,0)$. Deleting one vertex of $A$ leaves $(2,1)$. The Petersen graph is vertex-transitive, so both deletions are isomorphic to $P-v$. Lemma~\ref{lem:p6} records that $P$ is $P_6$-free and strong $C_4$-free, and both properties pass to induced subgraphs. From the table, $\lr(P)=3$ and $\lr(P-v)=8/3$. The graph $(2,0)$ has $\lr=5/2$.

\emph{Case 2: an induced five-cycle and no induced six-cycle.}
Write $C=c_0\cdots c_4c_0$. Every vertex at distance one from $C$ has a singleton or consecutive-pair trace: any other pair of contacts lies at distance two on $C$ and forms a $4$-cycle. If a vertex $x$ were at distance two, with a neighbour $a$ in the first layer, the path $x,a$ followed by four consecutive cycle vertices, beginning at one contact of $a$ and omitting the second contact if $a$ has one, would be an induced $P_6$. Thus $C$ dominates $G$.

Write $h$ for the number of vertices with singleton trace and $j$ for the number with a consecutive-pair trace. At most one vertex carries each of the five pair traces, since two would have two common neighbours, so $j\le 5$. The edges outside $C$ are of the following kinds.
\begin{itemize}
\item Two singleton vertices with different supports are nonadjacent. Neighbouring supports give a $4$-cycle. Supports at distance two give an induced $6$-cycle along the longer arc of $C$. Edges within one singleton class form a matching, because two neighbours in that class would be two common neighbours of the supporting cycle vertex. Let $m$ be the number of these edges.
\item No two pair vertices are adjacent. From their two distinct support edges one can choose adjacent cycle vertices, one from each edge, and those two vertices together with the pair vertices form a $4$-cycle.
\item A singleton supported at $c_i$ meets only a pair vertex supported on the opposite edge $\{c_{i+2},c_{i+3}\}$. Every other pair edge contains a neighbour of $c_i$ and yields a $4$-cycle. That pair vertex meets at most one vertex of the singleton class, again by the common-neighbour bound. Let $b$ be the number of singleton-to-pair edges, so $b\le\min(h,j)$.
\end{itemize}
Consequently $n=5+h+j$ and $e=5+h+2j+m+b$. The triangles are the $j$ pair triangles and the $m$ matching triangles, so $T=j+m$ and
\[
S=10+2h+j-m+2b.
\]
Therefore
\[
3S-7n=-5-h-4j-3m+6b.
\]
The inequality $6\min(h,j)\le h+5j$ holds because $6h\le h+5j$ when $h\le j$ and $6j\le h+5j$ when $j\le h$. Hence $6b\le h+5j$, and
\[
3S-7n\le -5-3m+j\le -5+j\le 0,
\]
where the last step is $j\le 5$. Thus $\lr\le 7/3<5/2$.

\emph{Case 3: no holes.}
The graph is chordal, and every nonempty induced subgraph has a simplicial vertex. The neighbourhood of a simplicial vertex is a clique. The degree is at most two: four vertices of a clique contain a $4$-cycle. A simplicial vertex of a connected chordal graph on at least two vertices is not a cut vertex, and the graph left by deleting it is chordal and strong $C_4$-free. In the graph present at the deletion, the loss is $L(v)=2d(v)-3t(v)$. Degree one gives loss $2$. Degree two, with the two neighbours adjacent, gives one triangle and loss $1$. Deleting vertices until a single vertex remains, one has $S\le 2(n-1)$. Thus $\lr\le 2(1-1/n)<5/2$.

The three cases give the proposition. The only graphs with $\lr>5/2$ are $P$ and $P-v$, arising in Case~1. Every other graph satisfies $\lr\le 8/3$, and the value $8/3$ occurs only for $P-v$.
\end{proof}

\begin{remark}\label{rem:km}
Karthick and Maffray give a structural decomposition for graphs with no induced $P_6$ and no $C_4$, and they discuss Petersen blow-ups and clique cutsets~\cite{km}. Their class is larger than the strong $C_4$-free class of Proposition~\ref{prop:gap}. A bound on $S$ for the pieces of such a decomposition still requires a gluing argument. We cite that work only as a comparison of scope. Priority of the gap relative to the literature would need a separate check, and none is claimed here.
\end{remark}

\section{Radius four}\label{sec:r4}

\begin{lemma}\label{lem:c8}
Let $G$ be connected, strong $C_4$-free and $P_8$-free. If an induced $8$-cycle dominates $G$ and $r(G)\ge 4$, then $G=C_8$.
\end{lemma}

\begin{proof}
Write $C=c_0c_1\cdots c_7c_0$, with indices modulo~$8$. For an outside vertex $y$ the trace $A_y=N(y)\cap V(C)$ is nonempty.

The trace $A_y$ is not contained in any three consecutive vertices of $C$. Suppose $A_y\subseteq\{c_i,c_{i+1},c_{i+2}\}$. The ends $c_i$ and $c_{i+2}$ are not both in $A_y$, since $y,c_i,c_{i+1},c_{i+2}$ would be a $4$-cycle. Thus $A_y$ is a singleton or a consecutive pair. A singleton $\{c_i\}$ gives the induced path
\[
y,c_i,c_{i+1},c_{i+2},c_{i+3},c_{i+4},c_{i+5},c_{i+6}.
\]
A consecutive pair $\{c_i,c_{i+1}\}$ gives the induced path
\[
y,c_i,c_{i-1},c_{i-2},c_{i-3},c_{i-4},c_{i-5},c_{i-6},
\]
in which the omitted contact $c_{i+1}$ does not lie on the path. Either path is an induced $P_8$.

Fix a cycle vertex, say $c_0$. The vertices at cycle distance greater than two from $c_0$ are the three consecutive vertices $c_3,c_4,c_5$. The trace of any outside vertex meets the complementary set $\{c_7,c_0,c_1,c_2,c_6\}$, so every outside vertex is within distance three of $c_0$. The same holds with $c_0$ replaced by any vertex of $C$.

Suppose $w$ lies outside $C$, and write $A=N(w)\cap V(C)$. No two vertices of $A$ are at cycle distance two, for they would form a $4$-cycle with $w$. A set of vertices of $C$ in which every two distinct vertices are consecutive has size at most two, and every set of size at most two that is a singleton or a consecutive pair is contained in a triple of consecutive vertices. Both possibilities have been excluded. Therefore $A$ contains two vertices at cycle distance three or four. Rotating, take them to be $c_0$ and $c_d$ with $d\in\{3,4\}$.

The antipode of $c_0$ is $c_4$. If $d=4$, the path $c_0wc_4$ has length two. If $d=3$, then $c_3\sim c_4$, so $c_0wc_3c_4$ has length three. Every other vertex of $C$ is already at cycle distance at most three from $c_0$. Every outside vertex is within distance three of $c_0$. Thus the eccentricity of $c_0$ is at most three, contradicting $r(G)\ge 4$.

Hence $C$ has no vertex outside it, and $G=C_8$.
\end{proof}

\begin{remark}\label{rem:c8enum}
The proof of Lemma~\ref{lem:c8} does not use a machine enumeration. An independent count, with the labelled cycle fixed, finds $32$ admissible nonempty traces and $790$ configurations on those traces, including the bare cycle. By order $8,\dots,13$ the configuration counts are $1,32,262,384,107,4$. There are $73$ unrooted isomorphism classes, and only the bare cycle has radius at least four. That count is not a hypothesis of Theorem~\ref{thm:r4}.
\end{remark}

\begin{theorem}\label{thm:r4}
Let $G$ be connected and strong $C_4$-free with $r(G)=4$.
\begin{enumerate}
\item If $\pp(G)=7$, then $S(G)\le 2n$, so $\lr(G)\le 2$.
\item If $\pp(G)\ge 8$ and $\lr(G)\le 4$, then $\pp(G)\ge r(G)+\lr(G)$.
\end{enumerate}
In particular $\RAD(4)$ holds: every such graph with $\pp(G)=7$ satisfies $\lr(G)\le 3$.
\end{theorem}

\begin{proof}
(2) is immediate: $\pp(G)\ge 8=r(G)+4\ge r(G)+\lr(G)$.

For (1), Theorem~\ref{thm:ess} is not required, but it is compatible: $\pp(G)\ge 7$. Apply Theorem~\ref{thm:cs} with $k=8$. There is a connected dominating set $X$ such that $G[X]$ is $P_6$-free or $G[X]\cong C_8$.

If $G[X]\cong C_8$, Lemma~\ref{lem:c8} gives $G=C_8$. Then $S=2n$ and $\lr=2$.

If $G[X]$ is $P_6$-free, apply Lemma~\ref{lem:ecc2} inside $G[X]$. A vertex of eccentricity at most two in $G[X]$ has eccentricity at most three in $G$, because $X$ dominates $G$, contradicting $r(G)=4$. Therefore $G[X]=C_6$. Let $C=v_0\cdots v_5$ be this dominating induced $6$-cycle, indices modulo~$6$, and let $W=V(G)\setminus V(C)$.

\emph{Traces.} As in the proof of Lemma~\ref{lem:c6}, strong $C_4$-freeness allows only three kinds of trace $N(w)\cap C$: a single vertex, a consecutive pair $\{v_i,v_{i+1}\}$, or an antipodal pair. If some $w$ has trace $\{v_i,v_{i+3}\}$, then $v_i$ is within distance two of every vertex of $C$, hence within distance three of every vertex of $G$, a contradiction. So there are no antipodal traces. Let $C_i$ be the set of vertices with trace $\{v_i\}$.

\emph{Witnesses.} Fix $i$. Since $r(G)=4$, some vertex is at distance at least four from $v_{i+3}$. Every vertex of $C$ is within distance three of $v_{i+3}$, and a vertex $w\in W$ is at distance at most $1+\min\{d_C(v_{i+3},x):x\in N(w)\cap C\}$ from $v_{i+3}$, which is at most three unless the trace of $w$ is $\{v_i\}$. So there is $w_i\in C_i$ at distance four from $v_{i+3}$. Every neighbour $z\in W$ of $w_i$ lies in $C_i$: if $z$ had a neighbour among $v_{i+2},v_{i+3},v_{i+4}$, then $d(w_i,v_{i+3})\le 3$; if $z$ had the neighbour $v_{i-1}$ or $v_{i+1}$, then $w_i,v_i,v_{i\pm 1},z$ would be a $4$-cycle. So $N(w_i)\subseteq\{v_i\}\cup C_i$.

\emph{Edges in $W$.} Let $x,y\in W$ be adjacent.
\begin{itemize}
\item $x\in C_0$, $y\in C_{\pm 1}$: $x,v_0,v_{\pm 1},y$ is a $4$-cycle.
\item $x\in C_0$, $y\in C_2$: $w_5,v_5,v_0,x,y,v_2,v_3,w_3$ is an induced $P_8$.
\item $x\in C_0$, $y\in C_3$: $w_1,v_1,v_0,x,y,v_3,v_4,w_4$ is an induced $P_8$. By symmetry, vertices of $C_i$ and $C_j$ with $i\ne j$ are never adjacent.
\item $x$ with trace $\{v_0,v_1\}$ and $y\in C_j$: for $j\in\{0,1\}$ the vertices $v_0,v_1,x,y$ span a $4$-cycle; for $j=5$ (respectively $j=2$) the cycle $x,v_0,v_5,y$ (respectively $x,v_1,v_2,y$) is a $4$-cycle; for $j=3$, $w_5,v_5,v_0,x,y,v_3,v_2,w_2$ is an induced $P_8$, and $j=4$ is symmetric.
\item $x,y$ with consecutive-pair traces $\{v_0,v_1\}$ and $\{v_j,v_{j+1}\}$: $j=0$ gives two common neighbours; $j=\pm 1$ and $j=\pm 2$ give $4$-cycles as above; $j=3$ gives the induced $P_8$ $w_2,v_2,v_1,x,y,v_4,v_5,w_5$.
\end{itemize}
In each displayed path the non-edges follow from the traces and from $N(w_i)\subseteq\{v_i\}\cup C_i$. So every edge inside $W$ joins two vertices of the same class $C_i$, and these edges form a matching.

\emph{Mass.} Let $a$ be the number of singleton-trace vertices, let $b$ be the number of consecutive-pair vertices, and let $m$ be the number of edges inside $W$. Then
\[
n=6+a+b,\qquad e=6+a+2b+m,\qquad T=b+m,
\]
so $S=2e-3T=2n-b-m\le 2n$, and $\lr\le 2$.
\end{proof}

\begin{theorem}[Theorem A]\label{thm:A}
Let $G$ be connected and strong $C_4$-free with $n\ge 2$. Then $\pp(G)\ge r(G)+\lr(G)$ whenever $\lr(G)\le 3$, whenever $r(G)\le 2$ and $\lr(G)\le 4$, whenever $r(G)=4$ and $\lr(G)\le 4$, and whenever $r(G)\ge 5$ and $\lr(G)\le 4$. If $r(G)=4$ and $\pp(G)=7$, then $\lr(G)\le 2$.
\end{theorem}

\begin{proof}
Write $r=r(G)$ and $c=\lceil\lr(G)\rceil$. The last sentence is Theorem~\ref{thm:r4}(1). For the inequality we need $\pp\ge r+c$ in the listed ranges.
\begin{itemize}
\item $r=1$: Lemma~\ref{lem:r1}.
\item $r=2$, $c\le 2$: Lemma~\ref{lem:r2}.
\item $r=2$, $c=3$: if $\pp\le 4$, then Lemma~\ref{lem:p5} gives $\lr\le 2$, a contradiction. So $\pp\ge 5$.
\item $r=2$, $c=4$: if $\pp\le 5$, then Lemma~\ref{lem:p6} gives $\lr\le 3$, a contradiction. So $\pp\ge 6$.
\item $r=3$, $c\le 2$: Theorem~\ref{thm:ess} gives $\pp\ge 5$.
\item $r=3$, $c=3$: if $\pp\le 5$, then by Lemma~\ref{lem:ecc2} and $r=3$ we have $G=C_6$, with $\lr=2$, a contradiction. So $\pp\ge 6$. Lemma~\ref{lem:p6} alone would give only $\lr\le 3$ for $\pp\le 5$, which does not rule out $c=3$.
\item $r\ge 4$, $c\le 3$: Theorem~\ref{thm:ess} gives $\pp\ge 2r-1\ge r+3$.
\item $r=4$ and $\lr\le 4$: Theorem~\ref{thm:ess} gives $\pp\ge 7$. If $\pp=7$, Theorem~\ref{thm:r4} gives $\lr\le 2$, so $\pp\ge r+\lr$. If $\pp\ge 8$, Theorem~\ref{thm:r4}(2) gives the same inequality.
\item $r\ge 5$ and $\lr\le 4$: Theorem~\ref{thm:ess} gives $\pp\ge 2r-1\ge r+4\ge r+c$.
\end{itemize}
\end{proof}

Equality in REAL-133 occurs for $C_5$ ($r=2$, $\lr=2$, $\pp=4$), the Petersen graph ($r=2$, $\lr=3$, $\pp=5$) and $C_6$ ($r=3$, $\lr=2$, $\pp=5$). We do not characterise all equality cases.

\section{The reduction and the residual cases}\label{sec:band}

For integers $K\ge 3$ and $r\ge 4$ consider the following statements.
\begin{itemize}
\item[$\GEN_K$:] every connected strong $C_4$-free graph $G$ with $\pp(G)\le 2K$ satisfies $S(G)\le Kn$.
\item[$\RAD(r)$:] every connected strong $C_4$-free graph with $r(G)=r$ and $\pp(G)=2r-1$ satisfies $\lr(G)\le r-1$.
\end{itemize}
Theorem~\ref{thm:r4} is the case $r=4$ of $\RAD(r)$, in the stronger form $\lr\le 2$.

\begin{theorem}[Theorem B]\label{thm:B}
Let $G$ be a connected strong $C_4$-free graph that violates Conjecture~\ref{conj:133}. Then $\lr(G)>3$. Put $K=\lceil\lr(G)\rceil-1\ge 3$. Then either
\begin{enumerate}
\item[(a)] $r(G)\le K$ and $\pp(G)\le r(G)+K\le 2K$, so $G$ violates $\GEN_K$; or
\item[(b)] $r(G)=K+1\ge 4$ and $\pp(G)=2r(G)-1$, and $\lr(G)>r(G)-1$, so $G$ violates $\RAD(K+1)$.
\end{enumerate}
\end{theorem}

\begin{proof}
Write $r=r(G)$, $\pp=\pp(G)$ and $c=\lceil\lr(G)\rceil=K+1$. Then $\lr>3$ by Theorem~\ref{thm:A}. A violation means $\pp\le r+c-1=r+K$. Theorem~\ref{thm:ess} gives $2r-1\le\pp\le r+K$, so $r\le K+1$. If $r\le K$, then $\pp\le r+K\le 2K$, and $S>Kn$ because $\lr>K$. If $r=K+1$, then $2r-1\le\pp\le 2K+1=2r-1$, and $\lr>K=r-1$.
\end{proof}

Since $\GEN_K$ ignores the radius, it is stronger than alternative~(a). The statement $\GEN_K$ might fail for large $K$ even if REAL-133 holds; see Remark~\ref{rem:moore}.

\begin{proposition}\label{prop:theta}
For $k\ge 3$ let $\theta_k$ consist of two vertices $x,y$ joined by $k$ internally disjoint paths $xa_ib_iy$. Then $\theta_k$ is strong $C_4$-free, $n=2k+2$, $r=3$, $\pp=6$ and $S=6k=3n-6$. Thus $\pp-r-\lr=6/n\to 0$.
\end{proposition}

\begin{proof}
Every cycle consists of two complete branches and so has length $6$; in particular there is no $4$-cycle. There are no triangles, so $S=2e=6k$. Every vertex has eccentricity $3$: $d(x,y)=3$ and $d(a_i,b_j)=3$ for $i\ne j$. The path $a_jxa_ib_iyb_l$ (with $j,i,l$ distinct) is induced. Conversely, let $P$ be an induced path. If $P$ avoids $x$ and $y$, it lies inside one edge $a_ib_i$. If $P$ contains $x$ but not $y$, then $x$ has at most two neighbours on $P$, each followed on $P$ by at most its $b$-vertex, so $|P|\le 5$. If $P$ contains $x$ and $y$, then every $a_j$ on $P$ is adjacent on $P$ to $x$ and every $b_j$ to $y$, at most one branch lies entirely on $P$, and so $P$ consists of $x$, $y$, at most one full branch and at most one further neighbour of each of $x$ and $y$. Hence $|P|\le 6$.
\end{proof}

Thus no argument with a uniform positive margin in $\lr$ can prove $\GEN_3$, already among graphs of radius~$3$.

\section{The rooted bound of order three}\label{sec:root3}

For a vertex $s$ of a connected graph $F$ let $\pp_s(F)$ be the largest order of an induced path starting at $s$. Write $D=d_F(s)$ and $m=e(F[N(s)])$.

\begin{definition}\label{def:root}
For an integer $q\ge 3$, the rooted bound $\ROOT(q)$ asserts that every connected strong $C_4$-free graph $F$ and every vertex $s$ with $\pp_s(F)\le q$ satisfy
\[
S(F)\le(q-1)(|F|-1)-(q-3)D-m.
\]
\end{definition}

The hypothesis $\pp_s(F)\le q$ belongs in the statement. Without it the displayed inequality is false: if $q=3$ and $F=C_5$, then $S(F)=10$ and $m=0$, while $2(|F|-1)=8$.

\begin{lemma}\label{lem:rootedp4}
$\ROOT(3)$ holds. Explicitly, if $F$ is connected and strong $C_4$-free and $s$ is a vertex at which no induced $P_4$ starts, then $S(F)\le 2(|F|-1)-m$.
\end{lemma}

\begin{proof}
A vertex at distance $3$ from $s$ would end a geodesic, hence an induced $P_4$, starting at $s$. So $V=\{s\}\cup N_1\cup N_2$ with $N_i$ the $i$-th distance layer. Each $y\in N_2$ has a unique parent $a\in N_1$. A neighbour $z\in N_2$ of $y$ with a different parent would give the induced path $sayz$. So $y$ has at most one neighbour in $N_2$, a sibling (two siblings would be common neighbours of $a$ and $y$), and $N(y)$ is a clique; thus $\lambda(y)=1$. For $a\in N_1$ we have $N(a)\subseteq\{s,\bar a\}\cup C_a$, where $C_a$ is the set of children of $a$ and $\bar a$ is the neighbour of $a$ in $N_1$, if any. Since $s$ and $\bar a$ lie in one unit, $\lambda(a)\le 1+|C_a|$. Finally $\lambda(s)=|N_1|-m$. Summing, $S\le(|N_1|-m)+(|N_1|+|N_2|)+|N_2|=2(|F|-1)-m$.
\end{proof}

For $q\ge 4$ the bound $\ROOT(q)$ is not proved for arbitrary strong $C_4$-free graphs. Section~\ref{sec:notclaimed} records the precise limitation. The next section proves every index inside a restricted class.

\section{Rooted bounds for every index}\label{sec:blocks}

Let $\mathcal B$ be the class of connected strong $C_4$-free graphs in which every block is bipartite or a cycle. Edges are blocks, and cycles may be odd. The class contains every connected strong $C_4$-free bipartite graph and every connected strong $C_4$-free cactus. The block restriction is necessary for a uniform rooted bound: $K_{2,3}$ rooted in its two-vertex part has $\pp_s=3$ and $S=12>2(n-1)-m=8$.

A rooted graph $(G,s)$ with $\pp_s(G)=q$ and $\mathrm{ecc}(s)=q-1$ is called \emph{terminal}. For a terminal instance put $R=n-q+1-D\ge 0$, the number of vertices outside $N[s]$ and outside a root-to-farthest-vertex geodesic. Set $c_2=0$, $c_3=2$, $c_4=3$ and $c_q=2\lfloor(q-1)/2\rfloor$ for $q\ge 5$.

\begin{theorem}\label{thm:blocks}
\begin{enumerate}
\item Every terminal instance in $\mathcal B$ satisfies $S(G)\le 2(D+q-2)-m+c_q R$.
\item Every $F\in\mathcal B$ with a vertex $s$ such that $\pp_s(F)\le q$ satisfies
\[
S(F)\le c_q(n-1)-(c_q-2)D-m.
\]
In particular $\ROOT(q)$ holds in $\mathcal B$ for every $q\ge 3$. If every block is bipartite or a triangle, one may take $c_q=2\lfloor(q-1)/2\rfloor$ for all $q\ge 3$.
\end{enumerate}
\end{theorem}

\begin{proof}
\emph{Two terminal facts.} An endpoint $x$ of a longest rooted induced path is not a cut vertex: a neighbour of $x$ in another component of $G-x$ would extend the path. If $z$ is at distance $q-1$ from $s$ and $v_0=s,\dots,v_{q-1}=z$ is a geodesic, any neighbour $y\ne v_{q-2}$ of $z$ must meet an earlier path vertex (else the rooted path extends); meeting $v_i$ with $i\le q-4$ shortens $d(s,z)$, and meeting $v_{q-3}$ gives a $C_4$; so $y\sim v_{q-2}$, and two such $y$ would give a $C_4$. Hence $L(z)=2d(z)-3t(z)\le 2$. Moreover, if $x\ne s,z$, $G-x$ is connected and $z$ survives, then $q-1\le d_{G-x}(s,z)\le\pp_s(G-x)-1\le q-1$, so the instance stays terminal with the same $q$.

\emph{Endpoint losses.} Let $x$ be the endpoint of a longest rooted path of order $q$. It lies in a single block. If that block is a triangle the loss $L(x)$ is $1$, if an edge $2$, if a cycle of length at least $5$ it is $4$. If it is bipartite, let $\ell$ be the order of the final path segment inside it. Block-tree separation shows that the neighbours of $x$ in this block can only be blocked by earlier path vertices in the same block; every neighbour other than the predecessor has a blocker in the colour class of $x$, the vertex two positions earlier cannot block (a $C_4$), and each position blocks at most one neighbour. So $d(x)\le\lfloor(\ell-1)/2\rfloor\le\lfloor(q-1)/2\rfloor$, there are no triangles at $x$, and $L(x)\le c_q$ for $q\ge 5$.

\emph{Bases.} For $q\le 4$ a nontrivial bipartite block would yield, by the same endpoint count, a rooted induced path of order at least $5$, so these instances are cacti. For a $C_4$-free cactus with $c_3$ triangle blocks and $c_L$ longer cycle blocks, $S=2(n-1)+2c_L-c_3$ and $m\le c_3$. For $q=3$ no longer cycle occurs, and (1) follows. For $q=4$ every longer cycle is a $C_5$ through $s$. A chosen length-$3$ geodesic uses at most one edge of each such $C_5$: if it used two edges and then left the cycle, replacing that two-edge arc by the other three-edge arc would give an induced rooted path of order $5$. So each $C_5$ contributes two vertices to $R$, $R\ge 2c_L$, and $[2(D+2)-m+3R]-S=R-2c_L+c_3-m\ge 0$. For $q=2$ the root is universal and $S=2D-m$.

\emph{Induction on $n$.} Let $E$ be the set of endpoints of rooted induced paths of order $q$ and $z$ a farthest vertex, so $z\in E$. If some $x\in E\setminus\{z\}$ exists, delete it: $x\notin N[s]$, $x$ is not a cut vertex, its loss is at most $c_q$, the instance remains terminal with the same $q$, $D$ and $m$, and $R$ drops by one. If $E=\{z\}$, delete $z$: the new rooted maximum is $q-1$, the eccentricity $q-2$, $R$ is unchanged and the loss is at most $2$; induction gives $S\le 2(D+q-3)-m+c_{q-1}R+2\le 2(D+q-2)-m+c_q R$. Deletion preserves membership in $\mathcal B$. This proves (1); the bipartite-or-triangle variant is identical.

(2) Attach a new path with $q-1$ edges at $s$. The result is terminal of index $q$, lies in $\mathcal B$, has order $n+q-1$, mass $S+2(q-1)$ and root degree $D+1$, and $m$ is unchanged. Substituting into (1) gives (2). Since $c_q\le q-1$ and $n-1-D\ge 0$,
\begin{align*}
c_q(n-1)-(c_q-2)D-m
&=c_q(n-1-D)+2D-m\\
&\le(q-1)(n-1-D)+2D-m\\
&=(q-1)(n-1)-(q-3)D-m,
\end{align*}
so $\ROOT(q)$ follows.
\end{proof}

A matching upper bound of shape $S\le(r-1)n-2r(r-3)$, for graphs in $\mathcal B$ of radius $r\ge 5$ and path order $2r-1$, has been proposed from a structural description of radius-extremal graphs. That description is not proved in this paper, and the radius bound in $\mathcal B$ is not claimed.

\begin{lemma}\label{lem:blockers}
Let $G$ be strong $C_4$-free and triangle-free, and let $x_0x_1\cdots x_k$ be an induced path that cannot be extended at $x_k$. If $d(x_k)\ge 2$, then $k\ge d(x_k)+1$. If moreover $G$ is bipartite, then $k\ge 2d(x_k)$.
\end{lemma}

\begin{proof}
Let $y\ne x_{k-1}$ be a neighbour of $x_k$. It is not on the path, and, since the path cannot be extended, $y$ has a neighbour $x_i$ with $i\le k-1$. Here $i\ne k-1$ (a triangle) and $i\ne k-2$ (the $4$-cycle $x_{k-2}x_{k-1}x_ky$), so $i\le k-3$. A vertex $x_i$ is adjacent to at most one neighbour of $x_k$. Hence the $d(x_k)-1\ge 1$ such neighbours need distinct blockers among $x_0,\dots,x_{k-3}$, and $k-2\ge d(x_k)-1$. If $G$ is bipartite, the blocker has the colour of $x_k$, so $i\equiv k\pmod 2$ and $i\le k-4$; there are $\lfloor k/2\rfloor-1$ such positions, and $\lfloor k/2\rfloor\ge d(x_k)$.
\end{proof}

\begin{proposition}\label{prop:regbip}
Every connected regular bipartite strong $C_4$-free graph satisfies $\GEN_K$ for every $K$ and satisfies REAL-133. Every Moore graph of diameter~$2$ and girth~$5$ satisfies REAL-133.
\end{proposition}

\begin{proof}
Let $G$ be $d$-regular. If $d=1$ then $G=K_2$, with $r=1$, $\lr=1$, $\pp=2$ and $S=n$, and both statements hold. Let $d\ge 2$. As $G$ is triangle-free, $\lambda\equiv d$ and $\lr=d$. Lemma~\ref{lem:blockers} applied to a longest induced path gives $\pp\ge 2d+1$, and Theorem~\ref{thm:ess} gives $\pp\ge 2r-1$; so $\pp\ge r+d$ both when $r\le d+1$ and when $r\ge d+1$. If $\pp\le 2K$, then $2d+1\le 2K$, so $S=dn<Kn$. A Moore graph of degree $d\ge 2$ is triangle-free and strong $C_4$-free with $r=2$ and $\lr=d$, and the first part of Lemma~\ref{lem:blockers} gives $\pp\ge d+2=r+\lr$.
\end{proof}

\begin{remark}\label{rem:moore}
A Moore graph of degree $57$, if it exists, has $n=57^2+1=3250$ and $S=57n$. It satisfies REAL-133 by Proposition~\ref{prop:regbip}, which only needs $\pp\ge 59$. The statement $\GEN_{57}$ holds for it with equality in the mass bound, but $\GEN_{56}$ asks for $\pp\ge 113$ because $S>56n$. We do not know whether such a graph would have $\pp\le 112$. If it did, it would refute $\GEN_{56}$ while satisfying REAL-133. Its existence would therefore not by itself separate the two statements, but it shows that a proof of every $\GEN_K$ is a different assertion from REAL-133.
\end{remark}

\section{Negative results}\label{sec:neg}

The examples in this section are explicit graphs.

\begin{example}\label{ex:radstar}
Let $s\ge 3$, $t\ge 3$ and $m\ge 1$ be integers. Let $G_{s,t,m}$ be the cycle $C_{2s}=c_0\cdots c_{2s-1}$ with a pendant path of $t$ edges at every cycle vertex, together with $m$ internally disjoint paths $c_0xzyc_1$. Then $n=2s(t+1)+3m$, $e=2s(t+1)+4m$ and $T=0$, so $e-n-T=m\ge 1$. For $(s,t,m)=(3,3,1)$ one has $n=27$, $e=28$, $r=6$ and $\pp=11$. These graphs satisfy REAL-133, since $\lr=2e/n<8/3$. They also show that $e=n+T$ fails for radius-extremal strong $C_4$-free graphs of every radius $r\ge 6$: here $r(G_{s,t,m})=s+t$ and $\pp(G_{s,t,m})=2(s+t)-1$.

Two vertices have at most one common neighbour, so $G$ is strong $C_4$-free and triangle-free. An added path of length four does not shorten distances among the older vertices, and $c_0$ remains within distance two of every vertex of such a path, so the eccentricity of $c_0$ is still $s+t$. A vertex $w$ of an added path meets the tip of the arm at $c_s$ only through $c_0$ or $c_1$, hence at distance at least $s+t$. Thus $r=s+t$. For the path order, let $B$ be the block formed by $C$ and the added paths. An induced path $P$ meets $B$ in a subpath $Q$ and can leave $B$ only at an end of $Q$ that is a cycle vertex, through at most $t$ arm vertices. The same case division as for a single ear shows $|P|\le 2s+2t-1$, using $t\ge 3$ and $s\ge 3$, and the cycle minus one vertex together with the arms at its two ends attains this order.
\end{example}

The last sentence of Example~\ref{ex:radstar} abbreviates a case division which we now record. If $Q$ avoids both $c_0$ and $c_1$, then $|Q|\le 2s-2$. If $Q$ contains exactly one of them, then either $Q$ lies on the cycle with $|Q|\le 2s-1$, or $|Q|\le 2s+2$ with at most one cycle end, or $|Q|\le 7$ with no cycle end. If $Q$ contains both, the same three shapes occur with $|Q|\le 2s-1$, $|Q|\le 2s+2$ and $|Q|\le 6$. Adding at most $t$ arm vertices at each cycle end of $Q$ gives $|P|\le\max\{2s-1+2t,\ 2s+2+t,\ 7\}=2s+2t-1$.

\begin{example}\label{ex:star5}
Forbidding induced paths on eleven vertices and requiring $L(v)\ge 6$ at every vertex does not forbid an induced star $K_{1,6}$. Two Petersen graphs sharing a vertex give a triangle-free strong $C_4$-free graph on $n=19$ vertices with $S=60$, minimum deletion loss $6$, $\pp=9$ and $\lambda=6$ at the shared vertex. For $b\ge 2$, gluing $b$ copies of the Petersen graph along a common edge gives a $2$-connected graph $B_b$ with $n=8b+2$, $S=28b+2$, minimum loss $6$, $\pp=9$ and $\lambda=2b+1$ at the two hubs. Thus singleton losses do not bound $\lambda$. These graphs are not counterexamples to $\GEN_5$: the eight non-hub vertices of one Petersen block have deletion loss $28<5\cdot 8+1$.
\end{example}

For an oriented edge $(u,v)$ of a connected strong $C_4$-free graph $G$, let $q$ be the maximum order of an induced path whose first two vertices are $u,v$ in that order, and let $E$ be the set of terminal vertices of the induced paths of that order. The endpoint inequality $\mathrm{EP}(q)$ asserts $D_G(E)\le q|E|$ for every such oriented edge of maximum order $q$. For a root $s$ with $\pp_s(G)=q$, if $E$ is the set of terminal vertices of the induced paths of order $q$ starting at $s$, the rooted inequality $\mathrm{EP}^+(q)$ asserts $D_G(E)\le(q-1)|E|$.

\begin{proposition}\label{prop:ep}
For every integer $q\ge 4$, $\mathrm{EP}(q)$ holds for all connected strong $C_4$-free graphs if and only if $\mathrm{EP}^+(q-1)$ does.
\end{proposition}

\begin{proof}
Both graphs below remain strong $C_4$-free, so $S=2e-3T$ and the deletion loss of a vertex set is determined by the edges and triangles that meet it.

First assume $\mathrm{EP}^+(q-1)$, and let $(u,v)$ be an oriented edge of maximum induced-path order $q$, with terminal set $E$. Delete $W=N[u]\setminus\{v\}$, and let $H$ be the component of $v$ in $G-W$. An induced path of $G$ that starts with $uv$ has no further vertex in $N[u]$, because the path is induced, so the remainder is an induced path of $H$ starting at $v$. Conversely, every induced path of $H$ starting at $v$ extends by $u$ to an induced path of $G$, since no vertex of $H$ other than $v$ is adjacent to $u$. Thus $\pp_v(H)=q-1$, and the terminal set in $H$ is still $E$.

Each $x\in E$ has at most one neighbour in $W$: any such neighbour is adjacent to $u$, and two of them would be two common neighbours of $u$ and $x$. Every neighbour of $x$ outside $H$ lies in $W$. Let $m_W$ be the number of edges between $E$ and $W$, so $m_W\le|E|$, and let $T_{\mathrm{extra}}$ be the number of triangles that meet $E$ and are not contained in $H$. Every such triangle uses one of those edges. Comparing edges and triangles,
\[
D_G(E)=D_H(E)+2m_W-3T_{\mathrm{extra}}\le D_H(E)+2|E|.
\]
The rooted inequality in $H$ is $D_H(E)\le(q-2)|E|$, because $\mathrm{EP}^+(q-1)$ contributes the factor $(q-1)-1$. Therefore $D_G(E)\le q|E|$.

Conversely, assume $\mathrm{EP}(q)$, and let $(H,v)$ be a connected strong $C_4$-free graph with $\pp_v(H)=q-1$ and terminal set $E$. Form $G$ by adding a vertex $u$ adjacent to $v$ and, for each $x\in E$, a new vertex $a_x$ adjacent exactly to $u$ and to $x$. Every terminal $x$ fails to be adjacent to $v$: an induced path of order $q-1\ge 3$ from $v$ to $x$ has no chord $vx$. The vertex $a_x$ has degree two. Any cycle through $a_x$ uses both edges $ua_x$ and $a_xx$, and then continues from $x$ to the other neighbour of $u$ on the cycle. That other neighbour is $v$ or some $a_y$. In the first subcase the $x$--$v$ path in $H$ has length at least two, so the cycle has length at least five. In the second the cycle has the form $u,a_x,x,\dots,y,a_y$ and again has length at least five. A cycle through $u$ that avoids every $a_x$ would need two neighbours of $u$ in $H$, but the only neighbour of $u$ in $H$ is $v$. Thus $G$ is strong $C_4$-free. No new triangle appears: the two neighbours of $a_x$ are nonadjacent, and the neighbours of $u$ are pairwise nonadjacent.

An induced path that starts with $uv$ contains no $a_x$. Otherwise the edge $ua_x$ would be a chord, since $a_x$ is not the successor of $u$ on that path. The induced $uv$-paths are therefore the paths $u$ followed by an induced path of $H$ starting at $v$. Their maximum order is $q$, and their terminal set is $E$. Exactly $|E|$ new edges meet $E$, namely the edges $a_xx$, and no new triangle meets $E$. Hence $D_G(E)=D_H(E)+2|E|$. The oriented inequality gives $D_G(E)\le q|E|$, so $D_H(E)\le(q-2)|E|$, which is $\mathrm{EP}^+(q-1)$.
\end{proof}

\begin{example}\label{ex:ep}
$\mathrm{EP}(9)$ fails. Let $G$ have vertices $0,\dots,14$ and adjacency lists
\[
\begin{array}{llll}
0{:}\ 4,8,11 & 1{:}\ 5,8,10 & 2{:}\ 6,8,9 & 3{:}\ 7,10\\
4{:}\ 0,9,10 & 5{:}\ 1,9,11 & 6{:}\ 2,10,11 & 7{:}\ 3,11\\
8{:}\ 0,1,2,12 & 9{:}\ 2,4,5 & 10{:}\ 1,3,4,6,13 & 11{:}\ 0,5,6,7,14\\
12{:}\ 8,13,14 & 13{:}\ 10,12 & 14{:}\ 11,12. &
\end{array}
\]
The graph is triangle-free and strong $C_4$-free, with $n=15$, $e=23$ and $S=46$. For the oriented edge $(12,8)$ the maximum order is $q=9$, attained only by
\[
12,8,2,9,4,10,3,7,11\qquad\text{and}\qquad 12,8,2,9,5,11,7,3,10.
\]
Thus $E=\{10,11\}$ and $D_G(E)=20>18$. The same computation gives a failure of $\mathrm{EP}^+(8)$ on the subgraph induced by $0,\dots,11$, rooted at $8$: $\pp_8=8$, $E=\{10,11\}$ and $D(E)=16>14$. The inequality $\mathrm{EP}(8)$ is not settled in this paper, nor is $\mathrm{EP}^+(6)$.
\end{example}

\begin{example}\label{ex:cb}
Let $k=\pp-r$. Call an induced subgraph $H$ of $G$ an \emph{eligible core} if it is proper, connected and dominating, and $\pp(H)\le\pp(G)-2$. The budget $\mathrm{CB}$ asks for an eligible core with
\[
S(G)-S(H)\le k\bigl(n-|H|\bigr)+|H|.
\]
Together with REAL-133 for $H$ and an induction on the order, the budget would imply REAL-133. The Petersen graph has $S=30$, $\pp=5$, $r=2$ and $k=3$, hence $\lr=3$. An eligible core has $\pp(H)\le 3$. A connected triangle-free strong $C_4$-free $P_4$-free graph is a star, by Lemma~\ref{lem:join} and the absence of triangles. In the Petersen graph a star with $k\le 3$ leaves has centre of degree~$3$. The vertices it dominates are the centre, its three neighbours, and at most two further neighbours of each leaf, at most $4+2k$ vertices in all. This is $10$ only for $k=3$, so the eligible cores are exactly the ten closed neighbourhoods $N[v]$. Each has $S(H)=6$, and $S(G)-S(H)=24>3\cdot 6+4=22$.

The unrestricted budget therefore fails. The example has $\lr=3$. It does not determine a cutoff such as $\lr>5$, and this paper contains no census of eligible cores.
\end{example}

\begin{example}\label{ex:heawood}
The degree, triangle and distance data of the Heawood graph ($n=14$, cubic, $T=0$, $r=3$, $S=42$, and $Z=28$ nonadjacent pairs with no common neighbour) satisfy the counting identities used above, including $\sum_v d(v)^2=n(n-1)+6T-2Z$, together with the lower bounds $\pp\ge 2r-1$ and $\pp\ge\min_v\lambda(v)+2$, if $\pp$ is replaced by the value $5$. But $5<r+\lr=6$. This system of scalar constraints does not imply REAL-133. The Heawood graph itself has $\pp=7$.
\end{example}

\begin{example}\label{ex:plane}
Let $F$ be the incidence graph of the projective plane $\mathrm{PG}(2,q)$, with $f=|F|=2(q^2+q+1)$ and $S(F)=(q+1)f$, and choose a root of $F$. Let $t=f-1$ and $b\ge 1$. Take a $6$-cycle $c_0\cdots c_5$, attach a pendant path with $t$ edges at each cycle vertex, and attach $b$ copies of $F$ at $c_0$ by identifying their roots with $c_0$. The result is strong $C_4$-free, because each new block meets the rest in a cut vertex, and
\[
n=6f+b(f-1),\qquad S=12f+b(q+1)f,\qquad r=t+3,\qquad \pp=2t+5=2r-1.
\]
Distances inside the flower are unchanged. Every vertex of an attached copy is within distance $f-1\le t$ of $c_0$, so each cycle vertex has eccentricity $t+3$, while a vertex $x$ of a copy has distance $d_F(x,c_0)+t+3$ from the tip of the arm at $c_3$. An induced path meets the cycle in at most five consecutive vertices, and each of its two ends outside the cycle contributes at most $t$ vertices. Mass is additive at cut vertices by Lemma~\ref{lem:cut}. For fixed $q$, as $b\to\infty$ one has $\lr\to(q+1)f/(f-1)$. Letting $q\to\infty$, with $r=f+2$, this limit grows like $\sqrt{r/2}$. For $q=5$ and $b=17$ one has $n=1409$, $r=64$, $\pp=127$ and $S=7068>5n$. These graphs satisfy $\RAD(r)$ for their own radius, and they show that $\lr$ is unbounded on the class $\pp=2r-1$. Any theorem $\RAD(r)$ must use a bound that depends on $r$.
\end{example}

\section{Relation to earlier work}\label{sec:related}

Three published theorems are quoted in Section~\ref{sec:prelim} and used as stated. Erd\H{o}s, Saks and S\'os proved that every connected graph has an induced path on at least $2r(G)-1$ vertices~\cite{ess}. With the bound $n\ge 2r$, this is the reduction of Theorem~\ref{thm:B}. Camby and Schaudt characterised the absence of a long induced path by a connected dominating set whose induced subgraph is a shorter path-free graph or a cycle~\cite{cs}. The radius-four argument applies that theorem and then classifies traces by hand. Seinsche's theorem on graphs with no induced $P_4$~\cite{seinsche} is used for the small-mass bounds.

The conjecture is item~133 of DeLaVina's list~\cite{wowii}. We follow the real-valued reading fixed in~\cite{fc133,fcpr}. The comparison with Chen's preprint~\cite{chen} is in the introduction. Remark~\ref{rem:km} compares the extremal gap of Proposition~\ref{prop:gap} with the structure theory of Karthick and Maffray~\cite{km} and claims no priority.

\section{Open problems}\label{sec:open}

Theorem~\ref{thm:A} and Theorem~\ref{thm:B} leave the following questions open. Unrestricted $\GEN_6$ and REAL-133 are among them.

\begin{enumerate}
\item \textbf{The cell $r=3$, $\pp=6$, $3<\lr\le 4$.} This is the only unsettled case of REAL-133 with $\lr\le 4$. It would follow from $\GEN_3$. Appendix~\ref{app:gen3} reduces a minimum counterexample to three local statements that are not proved here.
\item \textbf{The range $\lr>4$.} A counterexample falls under $\GEN_K$ or $\RAD(K+1)$ for $K=\lceil\lr\rceil-1\ge 4$. Neither family is proved for those indices. Inside $5<\lr\le 6$ the alternatives of Theorem~\ref{thm:B} specialise to the pairs $(r,\pp)=(4,9)$ and $(5,10)$, together with the cells already closed by Theorem~\ref{thm:A}. Those two pairs are not settled, for bipartite graphs or in general. Section~\ref{sec:notclaimed} explains why earlier partial arguments are not proofs.
\item \textbf{Rooted bounds past order three, and $\RAD(r)$ for $r\ge 5$.} $\ROOT(3)$ is Lemma~\ref{lem:rootedp4}. In the class $\mathcal B$, Theorem~\ref{thm:blocks} gives every index. For arbitrary strong $C_4$-free graphs, $\ROOT(q)$ is open for every $q\ge 4$, and $\RAD(r)$ is open for every $r\ge 5$. The case $\RAD(4)$ is Theorem~\ref{thm:r4}.
\item \textbf{Level $6$.} Let $\mathcal F_0$ be the class of connected triangle-free strong $C_4$-free graphs $G$ with $\pp(G)\in\{11,12\}$, minimum degree at least~$4$, and with every endpoint of every induced path of order $\pp(G)$ of degree at most~$7$. Appendix~\ref{app:delete} gives three proved implications about deletion. They do not show that every graph in $\mathcal F_0$ with $S>6n$ has a nonempty set $U$ of deletion loss at most $6|U|$, and they do not prove $S\le 6n$ for every triangle-free strong $C_4$-free graph.
\end{enumerate}

\begin{conjecture}\label{conj:uniform}
$\ROOT(q)$ holds for every $q\ge 3$. For every $r\ge 5$, every connected strong $C_4$-free graph of radius $r$ and path order $2r-1$ satisfies $S\le(r-1)n-2r(r-3)$, with equality only for $C_{2r}$.
\end{conjecture}

The case $q=3$ is Lemma~\ref{lem:rootedp4}, and every $q$ holds in $\mathcal B$ by Theorem~\ref{thm:blocks}. The radius bound is not claimed for $r\ge 5$.

\begin{conjecture}\label{conj:td}
If $F$ is connected and strong $C_4$-free, $\pp_s(F)=q$ and the eccentricity of $s$ is $q-1$, then
\[
S(F)\le(q-1)(n-1)-(q-3)D-m-(q-2)(q-3).
\]
\end{conjecture}

We do not derive $\ROOT(q)$ from Conjecture~\ref{conj:td} in this paper. The terminal bound in $\mathcal B$ is the first part of Theorem~\ref{thm:blocks}.

\begin{conjecture}\label{conj:cbhigh}
Every connected strong $C_4$-free graph $G$ that is not a cycle and has $\lr(G)>5$ has an eligible core $H$, in the sense of Example~\ref{ex:cb}, with
\[
S(G)-S(H)\le(\pp-r)\bigl(n-|H|\bigr)+|H|.
\]
\end{conjecture}

Example~\ref{ex:cb} shows that the budget without a restriction on $\lr$ is false, and that the Petersen graph is a counterexample with $\lr=3$. The example does not show that $5$ is the correct cutoff. If the conjecture, the range $\lr\le 5$, and an induction on the order were all available, REAL-133 would follow. The range $\lr\le 5$ is not a theorem, so this route is not a proof of REAL-133. Example~\ref{ex:heawood} shows that a proof cannot rely only on the scalar constraints listed there.

\section{Bounds not proved in this paper}\label{sec:notclaimed}

A finite search is not a proof in this article unless the search is replaced by a written argument. This repository does not contain a versioned archive of a checker, its inputs, and a completeness argument for the inequalities below. Stored verdict files, and hashes of those files, authenticate recorded marks. They do not establish the inequalities. None of the following is a hypothesis of a theorem in this paper.

\begin{enumerate}
\item $\GEN_3$ and $\GEN_4$, and $\GEN_K$ for every $K\ge 5$, are not proved. Appendix~\ref{app:gen3} is a conditional reduction for $\GEN_3$ only.
\item $\RAD(r)$ is not proved for any $r\ge 5$. The case $r=4$ is Theorem~\ref{thm:r4}.
\item $\ROOT(q)$ is not proved for any $q\ge 4$ outside the class $\mathcal B$. Any correct statement of $\ROOT(q)$ includes the hypothesis $\pp_s(F)\le q$, as in Definition~\ref{def:root}.
\item The bipartite form of $\GEN_5$ is not proved. In particular the pairs $(r,\pp)=(4,9)$ and $(5,10)$ are not closed for bipartite graphs, and they are not closed in general. No argument in this paper uses an absent classification of $2$-connected $P_6$-free blocks, an absent list of traces, or the unproved bipartite bound to declare either pair closed.
\item No structure theorem for connected strong $C_4$-free graphs of radius $r\ge 5$ and path order $2r-1$ is proved here. The radius bound in the class $\mathcal B$ that would depend on such a theorem is not claimed.
\item Example~\ref{ex:cb} does not count eligible cores. A reported count of cores of order at most~$12$ is not a result of this paper and is not used to justify the cutoff in Conjecture~\ref{conj:cbhigh}.
\item Unrestricted $\GEN_6$, with or without triangles, is open. REAL-133 is open.
\end{enumerate}

\section{A conditional reduction for \texorpdfstring{$\GEN_3$}{GEN3}}\label{app:gen3}

The three local statements named at the end of this appendix are not proved. With them, the hand arguments below would give $\GEN_3$. Without them, $\GEN_3$ remains open, and so does the cell $r=3$, $\pp=6$, $3<\lr\le 4$.

\begin{theorem}\label{thm:red}
Let $G$ be a counterexample to $\GEN_3$ of minimum order: $G$ is connected and strong $C_4$-free, $\pp(G)\le 6$, $S(G)\ge 3n+1$, and $n$ is minimum. Then $G$ is $2$-connected; $S(G)-S(G-U)\ge 3|U|+1$ for every nonempty proper $U\subset V(G)$; $2\lambda(v)-t(v)\ge 4$ for every $v$, so $\lambda(v)\ge 2$; and every vertex with $\lambda(v)=2$ has degree $2$ and nonadjacent neighbours, and no two vertices of degree $2$ are adjacent.
\end{theorem}

\begin{proof}
Since $S\le 2<3n$ for $n\le 2$, one has $n\ge 3$. Let $v$ be a cut vertex. For each component $C$ of $G-v$ let $\rho_C$ be the largest order of an induced path in $G[C\cup\{v\}]$ with end $v$. Two such paths from different components form an induced path on $\rho_C+\rho_{C'}-1\le 6$ vertices, so some component $C_1$ has $\rho_{C_1}\le 3$. Put $G_1=G[C_1\cup\{v\}]$ and $G_2=G-C_1$, with $n_i=|G_i|$, so $n=n_1+n_2-1$ and $n_1\ge 2$. Both graphs are connected, smaller than $G$, and $P_7$-free. Lemma~\ref{lem:rootedp4} gives $S(G_1)\le 2(n_1-1)$, and minimality gives $S(G_2)\le 3n_2$. By Lemma~\ref{lem:cut}, $S(G)\le 2(n_1-1)+3n_2=3n-(n_1-1)<3n$.

Every component $H$ of $G-U$, for $U$ nonempty and proper, is connected, smaller, and $P_7$-free, so $S(H)\le 3|H|$. Summing, $S(G-U)\le 3(n-|U|)$, and $S(G)\ge 3n+1$ gives the deletion bound. The case $U=\{v\}$ and Lemma~\ref{lem:loss} give $2\lambda(v)-t(v)\ge 4$. If $\lambda(v)=2$, then $t(v)=0$, so $d(v)=2$ and the neighbours are nonadjacent. Every degree-$2$ vertex has $\lambda=2$ and hence lies in no triangle. If $u\sim w$ both have degree $2$, with other neighbours $x$ and $y$, then $x\ne y$ because $t(u)=0$. The set $\{u\}$ is a unit of $x$ and $\{w\}$ is a unit of $y$, so Lemma~\ref{lem:loss} with $U=\{u,w\}$ gives loss $2+2+1+1=6<7$.
\end{proof}

The same mass bound for a disconnected graph is the sum of the bounds on its components. An isolated vertex has $S=0$. Call a graph \emph{admissible} if it is strong $C_4$-free and $P_7$-free, $L(v)\ge 4$ for every vertex, and no two adjacent vertices have degree~$2$. A minimum counterexample is admissible. In an admissible graph, $\lambda(v)=2$ forces $t(v)=0$ and $d(v)=2$. Write $B$ for the vertices of degree at least~$3$, the \emph{branch vertices}, and $Z$ for the degree-$2$ vertices, the \emph{ears}. Put $D=G[B]$, $d_D(v)=|N(v)\cap B|$ and $q(v)=|N(v)\cap Z|$. All triangles lie in $D$.

\begin{lemma}\label{lem:ears}
In an admissible graph, $q(v)\le 4$ for every vertex $v$.
\end{lemma}

\begin{proof}
Let $u_i$ be ears at $v$ with other neighbours $a_i$. The $a_i$ are distinct, nonadjacent to $v$, and of degree at least~$3$. Put $E_i=N(a_i)\setminus\{u_i\}$. Then $|E_i|\ge 2$ and $E_i\cap N[v]=\emptyset$.

If $a_i\not\sim a_j$, then $E_i=\{w,p\}$ and $E_j=\{w,q\}$ with $pq\in E$, $w\not\sim p$ and $w\not\sim q$. Indeed two common neighbours of $a_i$ and $a_j$ are forbidden, so $|E_i\cap E_j|\le 1$. Since $|E_i|\ge 2$, the private part $E_i\setminus E_j$ is nonempty, and likewise $E_j\setminus E_i$. If $p$ lies in the first private part and $q$ in the second with $p\not\sim q$, the path $pa_iu_ivu_ja_jq$ is induced. Thus the private parts are complete to each other. If one of them contained two vertices, those two vertices and one vertex of the other private part would form a $4$-cycle with the corresponding $a$-vertex. Both private parts therefore have size one, and $|E_i\cap E_j|=1$. An edge from the common vertex to a private vertex is another $4$-cycle. In particular $a_i$ and $a_j$ have degree~$3$ and lie in no triangle.

No three of the vertices $a_i$ are pairwise nonadjacent. Otherwise the sets $E_i$ meet pairwise in a single vertex. If the three intersections are the three pairs of a triple, the claim forces a triangle on that triple, and one $a_i$ together with two vertices of the triple forms a $4$-cycle. Otherwise $E_i=\{w,p_i\}$ with $p_1p_2p_3$ a triangle and $w$ nonadjacent to each $p_i$, and $vu_1a_1wa_2p_2p_3$ is an induced $P_7$.

Five ears are impossible. The graph $F$ they determine on $\{a_1,\dots,a_5\}$ has independence number at most two. If some vertex $a$ of $F$ is adjacent to the other four, those four induce a matching. The independence number forces that matching to be perfect, say $\{b_1b_2,c_1c_2\}$. Then $b_1\not\sim c_1$, so the preceding paragraph says that $b_1$ lies in no triangle, but $ab_1b_2$ is a triangle. Otherwise every vertex of $F$ has a non-neighbour, hence has degree~$3$ in $G$ and lies in no triangle. One neighbour of each such vertex is its ear, so $F$ has maximum degree at most two. It is triangle-free and has no $4$-cycle, and its independence number is at most two, so $F$ is a $5$-cycle. An ear at $v$ together with four vertices of that cycle is then an induced $P_7$.
\end{proof}

\begin{proposition}\label{prop:adm}
Let $G$ be admissible, with $B$, $Z$, $D$, $d_D$ and $q$ as above. Suppose that $d_D(v)\le 3$ for every $v$, that $d_D(v)=3$ implies $q(v)=0$, and that $d_D(v)=2$ with the two neighbours in $B$ nonadjacent implies $q(v)\le 2$. Then $S(G)\le 3n$.
\end{proposition}

\begin{proof}
Ears have nonadjacent neighbours and $Z$ is an independent set, so $S(G)=S(D)+4|Z|$. Since $|Z|=\frac12\sum_{v\in B}q(v)$ and $n=|B|+|Z|$,
\[
S-3n=\sum_{v\in B}\Bigl(d_D(v)-t_D(v)-3+\tfrac12 q(v)\Bigr).
\]
Lemma~\ref{lem:ears} gives $q(v)\le 4$. If $d_D(v)=2$ and $t_D(v)=0$, the two neighbours in $B$ are nonadjacent, for an edge between them would be a triangle. The three suppositions and the ear bound make every summand nonpositive: the summand is at most $-1$ when $(d_D,t_D)=(0,0)$, at most $0$ when $(d_D,t_D)=(1,0)$, at most $0$ when $(d_D,t_D)=(2,0)$, at most $0$ when $(d_D,t_D)=(2,1)$, and equal to $-t_D(v)\le 0$ when $d_D(v)=3$. The suppositions exclude $d_D(v)\ge 4$.
\end{proof}

The three suppositions in Proposition~\ref{prop:adm} are not proved in this paper. Theorem~\ref{thm:red} shows that a minimum counterexample to $\GEN_3$ is admissible, so those suppositions would imply $\GEN_3$. They do not, by themselves, make $\GEN_3$ a theorem.

\section{Deletion in the triangle-free class}\label{app:delete}

Let $\mathcal F$ be the class of finite triangle-free strong $C_4$-free graphs with $\pp\le 12$. Write $\varepsilon_6(G)=S(G)-6|G|$, and call $G$ \emph{positive} when $\varepsilon_6(G)>0$. A nonempty set $U\subseteq V(G)$ is \emph{$6$-cheap} when $D_G(U)\le 6|U|$.

\begin{lemma}\label{lem:tfmargin}
Let $G$ be triangle-free and strong $C_4$-free, and let $\varnothing\ne U\subseteq V(G)$. Then
\[
D_G(U)=2e\bigl(G[U]\bigr)+2e(U,V(G)\setminus U),
\]
and $\varepsilon_6(G)-\varepsilon_6(G-U)=D_G(U)-6|U|$. If $D_G(U)\le 6|U|$ and $\varepsilon_6(G)>0$, then $U\ne V(G)$ and $\varepsilon_6(G-U)\ge\varepsilon_6(G)>0$. If $K\ge 6$ is real and $D_G(U)\le 6|U|$, then $D_G(U)\le K|U|$.
\end{lemma}

\begin{proof}
By Lemma~\ref{lem:basic}, $\lambda(v)=d(v)$ and $S(G)=2e(G)$. The edges counted by $S(G)$ but not by $S(G-U)$ are exactly the edges that meet $U$, so $D_G(U)=2e(G[U])+2e(U,V(G)\setminus U)$. The identity for $\varepsilon_6$ is the definition. If $U=V(G)$, then $D_G(U)=S(G)$, hence $D_G(U)-6|U|=\varepsilon_6(G)$. Thus $\varepsilon_6(G)>0$ forbids $D_G(V(G))\le 6|V(G)|$. The inequality $D_G(U)\le 6|U|$ rearranges to $\varepsilon_6(G-U)\ge\varepsilon_6(G)$. For real $K\ge 6$ one has $6|U|\le K|U|$.
\end{proof}

\begin{lemma}\label{lem:degenerate}
Assume that every nonempty triangle-free strong $C_4$-free graph of minimum degree at least~$4$ contains an induced $P_{11}$. Then every triangle-free strong $C_4$-free graph with $\pp\le 10$ is $3$-degenerate, and therefore satisfies $e\le 3(n-1)$ and $S\le 6(n-1)$ whenever $n\ge 1$. In particular $S\le 6n$.
\end{lemma}

\begin{proof}
Let $H$ be a nonempty induced subgraph of a triangle-free strong $C_4$-free graph with $\pp\le 10$. Then $H$ is triangle-free and strong $C_4$-free, and $\pp(H)\le 10$, so $H$ has no induced $P_{11}$. The assumed hypothesis forces $\delta(H)\le 3$. Delete a vertex of degree at most~$3$ in the current graph and repeat. Each of the first $n-1$ deletions removes at most three edges, and the last vertex is isolated, so $e\le 3(n-1)$. Lemma~\ref{lem:basic}(3) gives $S=2e$.
\end{proof}

The path hypothesis of Lemma~\ref{lem:degenerate} is not proved in this paper.

\begin{proposition}\label{prop:Gequiv}
Let $\mathcal R\subseteq\mathcal F$. Suppose that every positive graph in $\mathcal F\setminus\mathcal R$ has a nonempty $6$-cheap set. Then the following are equivalent.
\begin{enumerate}
\item Every positive graph in $\mathcal R$ has a nonempty $6$-cheap set.
\item Every graph in $\mathcal F$ satisfies $S\le 6n$.
\end{enumerate}
The supposition is not a proof that any particular subclass is empty.
\end{proposition}

\begin{proof}
If (2) holds, then no graph in $\mathcal F$ is positive, so (1) holds vacuously. Conversely, assume (1) and the supposition, and suppose some graph in $\mathcal F$ is positive. Among all positive induced subgraphs of members of $\mathcal F$, choose $H$ of minimum order. Then $H\in\mathcal F$, and $H$ is connected. If $H$ had a nonempty $6$-cheap set $U$, Lemma~\ref{lem:tfmargin} would give a smaller positive induced subgraph $H-U$. Thus $H\in\mathcal R$, contradicting (1). Hence no graph in $\mathcal F$ is positive.
\end{proof}

Let $\mathcal F_0$ be the class defined in Section~\ref{sec:open}. Proposition~\ref{prop:Gequiv} does not show that $\mathcal F_0$ contains no positive graph.

\section{Ports of a distance-preserving dominating set}\label{app:ports}

\begin{lemma}\label{lem:core}
Let $G$ be triangle-free and strong $C_4$-free, and let $D\subseteq V(G)$ be a connected dominating set such that $d_G(x,y)=d_{G[D]}(x,y)$ for all $x,y\in D$. Write $H=V(G)\setminus D$.
\begin{enumerate}
\item Every vertex of $H$ has exactly one neighbour in $D$.
\item If $x,y\in H$ are adjacent, with neighbours $a$ and $b$ in $D$, then $a\ne b$, $a\not\sim b$, and $d_{G[D]}(a,b)\le 3$.
\end{enumerate}
\end{lemma}

\begin{proof}
Let $x\in H$. Since $D$ dominates and $x\notin D$, the vertex $x$ has a neighbour in $D$. If $a,b\in N(x)\cap D$ are distinct, then $d_G(a,b)\le 2$, hence $d_{G[D]}(a,b)\le 2$. Distance one gives the triangle $axb$. Distance two gives a $4$-cycle through $x$ and a length-$2$ path in $G[D]$.

For (2), if $a=b$ then $axy$ is a triangle, and if $a\sim b$ then $axyb$ is a $4$-cycle. The path $a,x,y,b$ has length three, so $d_{G[D]}(a,b)\le 3$.
\end{proof}

\begin{lemma}\label{lem:coreid}
Under the hypotheses of Lemma~\ref{lem:core}, the cut between $D$ and $H$ has exactly $|H|$ edges, and
\[
\varepsilon_6(G)=2e(G[D])-6|D|+2e(G[H])-4|H|.
\]
If in addition $e(G[H])\le 2|H|$, then $\varepsilon_6(G)\le 2e(G[D])-6|D|$. The inequality $\varepsilon_6(G)\le -2|D|$ does not follow from this identity alone. In particular the identity does not produce a $6$-cheap set in every triangle-free strong $C_4$-free graph of path order $11$ or $12$.
\end{lemma}

\begin{proof}
Each vertex of $H$ has exactly one neighbour in $D$. Lemma~\ref{lem:basic}(3) gives $S(G)=2\bigl(e(G[D])+e(G[H])+|H|\bigr)$. Subtract $6(|D|+|H|)$.
\end{proof}

If $D$ is a single vertex and the minimum degree of $G$ is at least~$4$, Lemma~\ref{lem:core} forces $G$ to be a star, whose leaves have degree one. Thus no one-vertex set satisfies the hypotheses of Lemma~\ref{lem:core} in a graph of minimum degree at least~$4$.
